请求示例代码
http.client
Requests
import http.client

conn = http.client.HTTPSConnection("")
payload = ''
headers = {
'Authorization': 'Bearer {{YOUR_API_KEY}}'
}
conn.request("GET", "/minimax/v1/query/video_generation?task_id=%7Btask_id%7D", payload, headers)
res = conn.getresponse()
data = res.read()
print(data.decode("utf-8"))
