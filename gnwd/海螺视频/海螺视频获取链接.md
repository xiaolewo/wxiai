import http.client

conn = http.client.HTTPSConnection("")
payload = ''
headers = {
'Authorization': 'Bearer {{YOUR_API_KEY}}'
}
conn.request("GET", "/minimax/v1/files/retrieve?file_id=302892389810432", payload, headers)
res = conn.getresponse()
data = res.read()
print(data.decode("utf-8"))
