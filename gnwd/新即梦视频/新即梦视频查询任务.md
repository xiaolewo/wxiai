string

请求示例代码
http.client
Requests
import http.client

conn = http.client.HTTPSConnection("")
payload = ''
headers = {
'Authorization': 'Bearer {{YOUR_API_KEY}}'
}
conn.request("GET", "/v2/videos/generations/", payload, headers)
res = conn.getresponse()
data = res.read()
print(data.decode("utf-8"))
